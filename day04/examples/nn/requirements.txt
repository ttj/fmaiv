# Dependencies for the Day-4 neural-network examples (robustness + MNIST).

# CPU-only -- no GPU. torch is PINNED: newer torch removed
# torch.onnx.symbolic_helper._quantized_ops, which the pinned auto_LiRPA still
# uses, so an unpinned (e.g. Colab's preinstalled) torch breaks BoundedModule.
# Install the small CPU wheels (~200 MB) instead of the ~2.5 GB CUDA build:
#
#     pip install torch==2.8.0 torchvision==0.23.0 --index-url https://download.pytorch.org/whl/cpu
#     pip install -r requirements.txt
#
# torchvision is used only by the MNIST notebook (06_day4_nn_mnist); robustness.py
# needs only torch. auto_LiRPA's maintained release is on GitHub (PyPI is stuck at
# an ancient 0.2/0.3), pinned to a commit for reproducibility. The notebooks
# force-install these same pins so they work on Colab too (overriding Colab's torch).
#
torch==2.8.0
torchvision==0.23.0
matplotlib
